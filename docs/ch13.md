## 13-4 `db.js`を書き換える

### やること

期限日をタイムゾーンのずれなく取り出すための書き方は、この講座でまだ習っていない技術（`TO_CHAR`によるSQL側での変換）を使います。**この部分は自分で考えず、次のコードにそのまま置き換えてください。**

<details>
<summary>db.jsの完成コードを表示する</summary>

`src/db.js`を、次の内容にすべて置き換える。

```js
// データベースへの接続と、SQLをまとめるファイル（第9章で説明）
const fs = require('fs');
const path = require('path');
const { Pool } = require('pg');

// DATABASE_URLが無いと、意味の分からない接続エラーになってしまうため、
// ここで気づけるようにする（第9章で.envを作り忘れた場合の案内）
if (!process.env.DATABASE_URL) {
  throw new Error('DATABASE_URLが設定されていません。.envを作成し、第9章の手順書を参考にNeonの接続文字列を貼り付けてください');
}

// 接続文字列はDATABASE_URLから読む。SSLの設定は接続文字列側で行うため、
// ここでは何も指定しない（CIのPostgreSQLはSSLなしのため）
const pool = new Pool({ connectionString: process.env.DATABASE_URL });

// db/schema.sqlを実行し、tasksテーブルがなければ作る。
// サーバー起動時とテストの開始時の両方から呼び出す（完成済みのコードとして提供）
async function initializeSchema() {
  const schemaPath = path.join(__dirname, '..', 'db', 'schema.sql');
  const schemaSql = fs.readFileSync(schemaPath, 'utf8');
  await pool.query(schemaSql);
}

// 期限日（due_date、DATE型）はそのまま取り出すと、サーバーのタイムゾーンの
// 影響で1日ずれることがある。TO_CHARで、あらかじめ「YYYY-MM-DD」の文字列に
// 変換してから受け取ることで、ずれを防ぐ（第13章で説明。完成済みのコードとして提供）
const TASK_COLUMNS = "id, title, done, created_at, TO_CHAR(due_date, 'YYYY-MM-DD') AS due_date";

// DBの行（created_atなどスネークケース）を、APIで使う形（createdAtなどキャメルケース）に変換する
function mapRowToTask(row) {
  const task = {
    id: row.id,
    title: row.title,
    done: row.done,
    createdAt: row.created_at,
    dueDate: row.due_date
  };
  return task;
}

// tasksテーブルの全件を取得する
async function getAllTasks() {
  const result = await pool.query('SELECT ' + TASK_COLUMNS + ' FROM tasks ORDER BY id');
  const taskList = [];
  for (const row of result.rows) {
    taskList.push(mapRowToTask(row));
  }
  return taskList;
}

// tasksテーブルに1件追加する。期限日が無ければnullを渡す
async function insertTask(title, dueDate) {
  const result = await pool.query(
    'INSERT INTO tasks (title, due_date) VALUES ($1, $2) RETURNING ' + TASK_COLUMNS,
    [title, dueDate]
  );
  return mapRowToTask(result.rows[0]);
}

// idで指定したタスクを更新する。見つからなければnullを返す。
// updatesにtitle・done・dueDateのいずれかだけがあれば、その項目だけを書き換え、
// 残りは今の値のまま変えない（第11章・第13章で追加。片方だけ送られても大丈夫にする）
async function updateTask(id, updates) {
  const currentResult = await pool.query('SELECT ' + TASK_COLUMNS + ' FROM tasks WHERE id = $1', [id]);
  if (currentResult.rows.length === 0) {
    return null;
  }
  const currentTask = currentResult.rows[0];

  let newTitle = currentTask.title;
  if (typeof updates.title === 'string') {
    newTitle = updates.title;
  }
  let newDone = currentTask.done;
  if (typeof updates.done === 'boolean') {
    newDone = updates.done;
  }
  let newDueDate = currentTask.due_date;
  if (typeof updates.dueDate === 'string') {
    newDueDate = updates.dueDate;
  }

  const result = await pool.query(
    'UPDATE tasks SET title = $1, done = $2, due_date = $3 WHERE id = $4 RETURNING ' + TASK_COLUMNS,
    [newTitle, newDone, newDueDate, id]
  );
  return mapRowToTask(result.rows[0]);
}

// idで指定したタスクを削除する。削除できたかどうかをtrue/falseで返す
async function deleteTaskById(id) {
  const result = await pool.query('DELETE FROM tasks WHERE id = $1', [id]);
  return result.rowCount > 0;
}

// テストの終わりにDB接続を閉じるために使う（第11章で説明）
async function closePool() {
  await pool.end();
}

// tasksテーブルの中身を全部消す。テストが他のテストの影響を受けないように、
// 各テストの前に呼び出す（テスト専用。第11章で説明）
async function clearAllTasks() {
  await pool.query('DELETE FROM tasks');
}

module.exports = {
  initializeSchema: initializeSchema,
  getAllTasks: getAllTasks,
  insertTask: insertTask,
  updateTask: updateTask,
  deleteTaskById: deleteTaskById,
  closePool: closePool,
  clearAllTasks: clearAllTasks
};
```

</details>

### 確認すること

- `getAllTasks`・`insertTask`・`updateTask`のそれぞれで、`SELECT`や`RETURNING`のあとに`*`ではなく`TASK_COLUMNS`が使われている（全部で4か所あります。`*`が残っていないか、ファイル内を検索して確認しましょう）
- `insertTask`・`updateTask`の引数に、`dueDate`（または`updates.dueDate`）が増えている

## 13-5 APIで期限日を受け取り、`db.js`の関数に渡す

### やること

`src/app.js`の`createTask`・`updateTask`で、期限日を受け取って`db.js`の関数に渡します。書き方は、既存の`title`・`done`の扱いと同じ形です。

<details>
<summary>ヒント1：方針</summary>

- 追加のとき：`req.body.dueDate`が文字列で送られていれば使い、無ければ`null`にする
- 更新のとき：`req.body.dueDate`が文字列で送られていれば「更新する項目」に含める（`done`・`title`と同じ扱い）
- 値の中身（日付の形式が正しいかなど）はチェックしません。「文字列かどうか」だけ確認します

</details>

<details>
<summary>ヒント2：変更するファイルと関数</summary>

`src/app.js`の`createTask`関数・`updateTask`関数

</details>

<details>
<summary>ヒント3：コードの一部</summary>

`createTask`関数の中、`db.insertTask(title)`を呼び出す前に追加する。

```js
// 期限日は送られていれば使い、無ければnull（第13章で追加）
let dueDate = null;
if (typeof req.body.dueDate === 'string') {
  dueDate = req.body.dueDate;
}
```

呼び出し部分を`db.insertTask(title, dueDate);`に書き換える。

`updateTask`関数の中、`titleIsSent`の判定の近くに追加する。

```js
// 期限日も、doneやtitleと同じように「送られていれば使う」（第13章で追加）
const dueDateIsSent = typeof req.body.dueDate === 'string';
```

`if (doneIsSent === false && titleIsSent === false)`の条件に`&& dueDateIsSent === false`を追加し、エラーメッセージも「完了状態（done）・タイトル（title）・期限日（dueDate）のいずれかを送ってください」に変える。

`updates`を組み立てている部分に追加する。

```js
if (dueDateIsSent) {
  updates.dueDate = req.body.dueDate;
}
```

**次のコードは、そのまま書き写してください（完成済みのコードとして提供）。** APIに直接おかしな日付文字列が送られたとき、PostgreSQLが返すエラー（`22007`）を、500ではなく400に変換する部分です。正規表現などの新しい技術が必要になるため、ここだけは自分で考えず提供します。`respondToError`関数の中に追加する。

```js
// PostgreSQLが「日付の形式が正しくない」と判断したときのエラーコード。
// 期限日におかしな値が直接送られてきた場合、500ではなく400で返す
// （第13章で追加。完成済みのコードとして提供）
if (error.code === '22007') {
  res.status(400).json({ error: '期限日の形式が正しくありません' });
  return;
}
```

`typeof error.status === 'number'`の判定の後、`console.error(error);`の前に追加してください。

</details>

### うまくいったときの結果

- REST Clientで期限日付きのタスクを追加すると、`dueDate`が`"2026-05-01"`のような`YYYY-MM-DD`形式で返る
- 期限日なしで追加すると、`dueDate`が`null`になる
- 期限日に直接おかしな文字列（例：`"invalid"`）を送ると、500ではなく400が返る

## 13-6 期限日のテストを追加する

### やること

`tests/tasks.test.js`に、期限日に関するテストを追加します。既存のテストを真似してください。

<details>
<summary>ヒント1：方針</summary>

POSTで期限日を送った場合・送らなかった場合・おかしな形式を送った場合の3パターンと、PUTで期限日だけ更新してもタイトルが変わらないことを確認するテストを書きます。

</details>

<details>
<summary>ヒント2：真似する既存のテスト</summary>

- 「POST /api/tasksは、タイトルを送ると201でタスクを追加する」
- 「PUT /api/tasks/:idで完了状態だけを更新しても、タイトルは変わらない」

</details>

<details>
<summary>ヒント3：コードの例</summary>

```js
test('POST /api/tasksは、期限日を送るとYYYY-MM-DD形式で返す', async function () {
  const response = await request(app)
    .post('/api/tasks')
    .send({ title: '期限日つきのタスク', dueDate: '2026-05-01' });
  assert.strictEqual(response.status, 201);
  assert.strictEqual(response.body.dueDate, '2026-05-01');
});

test('POST /api/tasksは、期限日を送らないとdueDateがnullになる', async function () {
  const response = await request(app)
    .post('/api/tasks')
    .send({ title: '期限日なしのタスク' });
  assert.strictEqual(response.status, 201);
  assert.strictEqual(response.body.dueDate, null);
});

test('POST /api/tasksは、おかしな形式の期限日を送ると400を返す', async function () {
  const response = await request(app)
    .post('/api/tasks')
    .send({ title: '期限日がおかしいタスク', dueDate: 'おかしな日付' });
  assert.strictEqual(response.status, 400);
});

test('PUT /api/tasks/:idは、期限日だけを更新しても、タイトルは変わらない', async function () {
  const createdResponse = await request(app)
    .post('/api/tasks')
    .send({ title: '元のタイトル', dueDate: '2026-05-01' });
  const taskId = createdResponse.body.id;

  const response = await request(app)
    .put('/api/tasks/' + taskId)
    .send({ dueDate: '2026-06-15' });
  assert.strictEqual(response.status, 200);
  assert.strictEqual(response.body.dueDate, '2026-06-15');
  assert.strictEqual(response.body.title, '元のタイトル');
});
```

</details>
